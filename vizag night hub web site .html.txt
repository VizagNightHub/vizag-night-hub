  <script>
    // PRE-DEFINED HOST CREDENTIALS
    const HOST_EMAIL = "vardhansai8116@gmail.com";
    const HOST_PASS = "Vardhan@8116";

    // APPLICATION STATE (Uses localStorage for browser persistence)
    let generatedOtp = null;
    let tempRegEmail = "";

    function initStorage() {
      if (!localStorage.getItem("vize_users")) {
        const initialUsers = {};
        initialUsers[HOST_EMAIL] = HOST_PASS;
        localStorage.setItem("vize_users", JSON.stringify(initialUsers));
      }
      if (!localStorage.getItem("vize_members")) {
        localStorage.setItem("vize_members", JSON.stringify([HOST_EMAIL]));
      }
      if (!localStorage.getItem("vize_messages")) {
        localStorage.setItem("vize_messages", JSON.stringify([
          { id: 1, sender: HOST_EMAIL, text: "Welcome to VIZE NIGHT HUB! Feel free to share messages, media, or your location.", media: null, location: null, time: "System" }
        ]));
      }
    }

    initStorage();

    // AUTH SWITCHING
    function toggleAuthMode(mode) {
      document.getElementById("loginSection").style.display = mode === 'login' ? 'block' : 'none';
      document.getElementById("registerSection").style.display = mode === 'register' ? 'block' : 'none';
      document.getElementById("otpSection").style.display = mode === 'otp' ? 'block' : 'none';
    }

    // OTP FLOW
    function sendOtp() {
      const email = document.getElementById("regEmail").value.trim().toLowerCase();
      if (!email || !email.includes("@")) {
        alert("Please enter a valid email address.");
        return;
      }
      tempRegEmail = email;
      generatedOtp = Math.floor(100000 + Math.random() * 900000).toString();
      
      // Simulated Email Dispatch for immediate client-side execution
      alert(`[VIZE SECURITY] An OTP has been dispatched to: ${email}\n\nYour Verification Code is: ${generatedOtp}`);
      toggleAuthMode('otp');
    }

    function verifyAndSetPassword() {
      const enteredOtp = document.getElementById("enteredOtp").value.trim();
      const newPassword = document.getElementById("newPassword").value.trim();

      if (enteredOtp !== generatedOtp) {
        alert("Invalid OTP code. Please try again.");
        return;
      }
      if (newPassword.length < 4) {
        alert("Password must be at least 4 characters.");
        return;
      }

      const users = JSON.parse(localStorage.getItem("vize_users"));
      users[tempRegEmail] = newPassword;
      localStorage.setItem("vize_users", JSON.stringify(users));

      // Add to member list if new
      const members = JSON.parse(localStorage.getItem("vize_members"));
      if (!members.includes(tempRegEmail)) {
        members.push(tempRegEmail);
        localStorage.setItem("vize_members", JSON.stringify(members));
      }

      alert("Account verified! You can now log in.");
      toggleAuthMode('login');
      document.getElementById("loginEmail").value = tempRegEmail;
    }

    // LOGIN LOGIC
    function handleLogin() {
      const email = document.getElementById("loginEmail").value.trim().toLowerCase();
      const pass = document.getElementById("loginPassword").value.trim();

      const users = JSON.parse(localStorage.getItem("vize_users"));

      if (users[email] && users[email] === pass) {
        sessionStorage.setItem("currentUser", email);
        launchApp(email);
      } else {
        alert("Invalid Email or Password. If you are new, please register first.");
      }
    }

    function launchApp(email) {
      document.getElementById("authBox").style.display = "none";
      document.getElementById("appBox").style.display = "grid";

      document.getElementById("myEmailLabel").innerText = email;
      document.getElementById("myAvatar").innerText = email[0].toUpperCase();
      
      const isHost = email === HOST_EMAIL;
      document.getElementById("myRoleLabel").innerText = isHost ? "Group Host (Admin)" : "Member";
      if (isHost) {
        document.getElementById("hostPrivilegeNotice").style.display = "block";
      }

      renderMembers();
      renderMessages();
    }

    function logout() {
      sessionStorage.removeItem("currentUser");
      window.location.reload();
    }

    // GROUP MEMBER MANAGEMENT (HOST ONLY)
    function renderMembers() {
      const members = JSON.parse(localStorage.getItem("vize_members")) || [];
      const current = sessionStorage.getItem("currentUser");
      const isHost = current === HOST_EMAIL;
      const listEl = document.getElementById("membersList");
      listEl.innerHTML = "";

      members.forEach(member => {
        const item = document.createElement("div");
        item.className = "member-item";
        
        const isMemberHost = member === HOST_EMAIL;
        
        item.innerHTML = `
          <div class="member-meta">
            <div class="avatar" style="${isMemberHost ? 'background: var(--host-badge);' : ''}">${member[0].toUpperCase()}</div>
            <div>
              <div class="member-email">${member}</div>
              ${isMemberHost ? '<span class="badge-host">HOST</span>' : ''}
            </div>
          </div>
          ${(isHost && !isMemberHost) ? `<button class="btn-kick" onclick="removeMember('${member}')" title="Remove Member"><i class="fa-solid fa-user-xmark"></i></button>` : ''}
        `;
        listEl.appendChild(item);
      });
    }

    function removeMember(email) {
      if (confirm(`Are you sure you want to kick ${email} from the group?`)) {
        let members = JSON.parse(localStorage.getItem("vize_members"));
        members = members.filter(m => m !== email);
        localStorage.setItem("vize_members", JSON.stringify(members));
        renderMembers();
      }
    }

    // MESSAGES LOGIC
    function renderMessages() {
      const messages = JSON.parse(localStorage.getItem("vize_messages")) || [];
      const currentUser = sessionStorage.getItem("currentUser");
      const isHost = currentUser === HOST_EMAIL;
      const container = document.getElementById("messagesContainer");
      container.innerHTML = "";

      messages.forEach(msg => {
        const isMe = msg.sender === currentUser;
        const msgEl = document.createElement("div");
        msgEl.className = `message-bubble ${isMe ? 'me' : 'other'}`;

        let mediaHtml = "";
        if (msg.media) {
          if (msg.media.type.startsWith("image")) {
            mediaHtml = `<img src="${msg.media.data}" alt="Uploaded file" />`;
          } else if (msg.media.type.startsWith("video")) {
            mediaHtml = `<video controls src="${msg.media.data}"></video>`;
          }
        }

        let locHtml = "";
        if (msg.location) {
          locHtml = `<p><i class="fa-solid fa-map-location-dot"></i> <a href="${msg.location}" target="_blank" style="color:#fff; text-decoration:underline;">View Live Location on Google Maps</a></p>`;
        }

        msgEl.innerHTML = `
          <span class="bubble-sender">${msg.sender} • ${msg.time}</span>
          <div class="bubble-content">
            ${msg.text ? `<p>${msg.text}</p>` : ''}
            ${mediaHtml}
            ${locHtml}
          </div>
          ${isHost ? `<div class="bubble-actions"><button class="btn-del-msg" onclick="deleteMessage(${msg.id})"><i class="fa-solid fa-trash"></i> Delete</button></div>` : ''}
        `;
        container.appendChild(msgEl);
      });

      container.scrollTop = container.scrollHeight;
    }

    function sendTextMessage() {
      const input = document.getElementById("chatInput");
      const text = input.value.trim();
      if (!text) return;

      addMessage({ text: text });
      input.value = "";
    }

    function handleFileUpload(event) {
      const file = event.target.files[0];
      if (!file) return;

      const reader = new FileReader();
      reader.onload = function(e) {
        addMessage({
          media: {
            type: file.type,
            data: e.target.result
          }
        });
      };
      reader.readAsDataURL(file);
    }

    function shareLiveLocation() {
      if (!navigator.geolocation) {
        alert("Geolocation is not supported by your browser.");
        return;
      }
      navigator.geolocation.getCurrentPosition(position => {
        const lat = position.coords.latitude;
        const lng = position.coords.longitude;
        const mapsUrl = `https://www.google.com/maps?q=${lat},${lng}`;
        addMessage({ location: mapsUrl });
      }, () => {
        alert("Unable to retrieve location. Please grant location permissions.");
      });
    }

    function addMessage(payload) {
      const messages = JSON.parse(localStorage.getItem("vize_messages")) || [];
      const user = sessionStorage.getItem("currentUser");
      const timeStr = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

      messages.push({
        id: Date.now(),
        sender: user,
        text: payload.text || null,
        media: payload.media || null,
        location: payload.location || null,
        time: timeStr
      });

      localStorage.setItem("vize_messages", JSON.stringify(messages));
      renderMessages();
    }

    function deleteMessage(id) {
      let messages = JSON.parse(localStorage.getItem("vize_messages"));
      messages = messages.filter(m => m.id !== id);
      localStorage.setItem("vize_messages", JSON.stringify(messages));
      renderMessages();
    }

    // QR & INVITE MODAL
    function openInviteModal() {
      const currentUrl = window.location.href;
      document.getElementById("inviteUrl").value = currentUrl;
      document.getElementById("qrcode").innerHTML = "";
      new QRCode(document.getElementById("qrcode"), {
        text: currentUrl,
        width: 160,
        height: 160
      });
      document.getElementById("inviteModal").style.display = "flex";
    }

    function closeInviteModal() {
      document.getElementById("inviteModal").style.display = "none";
    }

    function copyInviteLink() {
      const urlInput = document.getElementById("inviteUrl");
      urlInput.select();
      navigator.clipboard.writeText(urlInput.value);
      alert("Group invite link copied to clipboard!");
    }
